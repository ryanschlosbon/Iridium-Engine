# Tests link the engine module libraries; no production source is compiled here.
#
# ctest labels:
#   unit     (default) self-contained, no GPU
#   vulkan   needs a Vulkan device
#   process  spawns child processes (cmake -P drivers, re-invoked executables)
#   slow     takes more than ~5 s (in either configuration)
#
# Every test reads PROJECT_ROOT_DIR from the directory-level definition
# ("<source>/"). Several targets used to add PROJECT_ROOT_DIR="<source>" (no
# trailing slash) as well; the directory definition was emitted last and won, so
# those duplicate definitions were dead and are gone.

# --- behavioral contract test support (M7R R2.1) -------------------------------------
# SPIR-V reflection, glslc depfile reading, render-graph queries and a headless
# Vulkan device. Paths reach the tests through a generated header rather than
# compile definitions, so test targets keep sharing the PCH.
if(TARGET iridium_shaders)
    set(_iridium_compiled_shader_entries "")
    foreach(_spv IN LISTS IRIDIUM_SHADER_SPV_FILES)
        get_filename_component(_spv_name "${_spv}" NAME)
        string(APPEND _iridium_compiled_shader_entries "    \"${_spv_name}\",\n")
    endforeach()
    set(IRIDIUM_TEST_PARITY_SHADER_DIR "${CMAKE_CURRENT_BINARY_DIR}/shaders")
    file(MAKE_DIRECTORY "${IRIDIUM_TEST_PARITY_SHADER_DIR}")
    file(CONFIGURE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/support/IridiumTestPaths.h"
            CONTENT "// Generated by tests/CMakeLists.txt. Do not edit.
#pragma once
#define IRIDIUM_TEST_SHADER_BINARY_DIR \"@IRIDIUM_SHADER_SOURCE_DIR@\"
#define IRIDIUM_TEST_SHADER_DEPFILE_DIR \"@IRIDIUM_SHADER_DEPFILE_DIR@\"
#define IRIDIUM_TEST_PARITY_SHADER_DIR \"@IRIDIUM_TEST_PARITY_SHADER_DIR@\"
// Every SPIR-V module the iridium_shaders target produces (stale files in
// assets/shaders are ignored).
inline constexpr const char* kIridiumCompiledShaders[] = {
@_iridium_compiled_shader_entries@};
" @ONLY)

    add_library(iridium_test_support STATIC
            support/SpirvInspector.cpp
            support/ShaderDepfile.cpp)
    target_include_directories(iridium_test_support PUBLIC
            "${CMAKE_CURRENT_SOURCE_DIR}/support"
            "${CMAKE_CURRENT_BINARY_DIR}/support")
    target_link_libraries(iridium_test_support PUBLIC iridium_core)
    add_dependencies(iridium_test_support iridium_shaders)

    add_library(iridium_test_vulkan_support STATIC
            support/HeadlessVulkanDevice.cpp
            support/HeadlessComputeKernel.cpp)
    target_link_libraries(iridium_test_vulkan_support PUBLIC
            iridium_test_support Vulkan::Vulkan)

    # Parity kernels: tests/shaders/*.comp include production GLSL from
    # assets/shaders and compile into the build tree only.
    file(GLOB _iridium_parity_sources CONFIGURE_DEPENDS
            "${CMAKE_CURRENT_SOURCE_DIR}/shaders/*.comp")
    set(_iridium_parity_outputs)
    foreach(_parity_source IN LISTS _iridium_parity_sources)
        get_filename_component(_parity_name "${_parity_source}" NAME)
        string(REPLACE "." "_" _parity_base "${_parity_name}")
        set(_parity_output "${IRIDIUM_TEST_PARITY_SHADER_DIR}/${_parity_base}.spv")
        set(_parity_depfile "${IRIDIUM_TEST_PARITY_SHADER_DIR}/${_parity_base}.spv.d")
        add_custom_command(
                OUTPUT "${_parity_output}"
                COMMAND "${GLSLC_COMPILER}" -I "${IRIDIUM_SHADER_SOURCE_DIR}"
                        "${_parity_source}" -o "${_parity_output}"
                        -MD -MF "${_parity_depfile}"
                DEPENDS "${_parity_source}"
                DEPFILE "${_parity_depfile}"
                COMMENT "Compiling test kernel ${_parity_name}"
                VERBATIM)
        list(APPEND _iridium_parity_outputs "${_parity_output}")
    endforeach()
    add_custom_target(iridium_test_shaders DEPENDS ${_iridium_parity_outputs})
endif()

# --- explicit qualification oracles (M7R R2.5) --------------------------------------
# --validation no longer implies the CPU oracles; these runs keep device/CPU parity
# exercised on tracked engine-authored fixtures. The oracle flags exist only in
# IRIDIUM_QUALIFICATION=ON builds.
if(TARGET iridium_qualification)
foreach(_oracle_case IN ITEMS
        "point_shadow_contact_v1|assets/m7-local-shadow-device-manifest.v1.json"
        "m7_heterogeneous_shadow_warm_motion_v1|assets/m7-heterogeneous-shadow-admission-manifest.v1.json")
    string(REPLACE "|" ";" _oracle_parts "${_oracle_case}")
    list(GET _oracle_parts 0 _oracle_fixture)
    list(GET _oracle_parts 1 _oracle_manifest)
    set(_oracle_name "EngineOracle.shadow_indirect.${_oracle_fixture}")
    add_test(NAME ${_oracle_name}
            COMMAND ${CMAKE_COMMAND}
                -DENGINE=$<TARGET_FILE:IridiumEngine>
                -DCOOK=$<TARGET_FILE:IridiumCookAsset>
                -DSOURCE_ROOT=${PROJECT_SOURCE_DIR}
                -DWORK_DIR=${CMAKE_BINARY_DIR}/oracle-tests/${_oracle_fixture}
                -DMODEL=benchmarks/m5/directional-shadow-contact.gltf
                -DBENCHMARK=${_oracle_fixture}
                -DMANIFEST=${_oracle_manifest}
                -DORACLE_ARGS=--shadow-indirect-qualification-oracle
                -P ${CMAKE_CURRENT_SOURCE_DIR}/renderer/EngineOracleProcessTest.cmake
            WORKING_DIRECTORY ${CMAKE_BINARY_DIR})
    set_tests_properties(${_oracle_name} PROPERTIES
            LABELS "vulkan;process"
            ENVIRONMENT "VK_LOADER_LAYERS_DISABLE=~implicit~")
endforeach()
endif()
